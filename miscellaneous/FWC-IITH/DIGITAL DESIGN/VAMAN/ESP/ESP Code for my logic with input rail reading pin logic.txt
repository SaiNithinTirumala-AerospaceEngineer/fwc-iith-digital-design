#include <WiFi.h>
#include <WiFiUdp.h>
#include <ArduinoOTA.h>

// Replace with your network credentials
#ifndef STASSID
#define STASSID "Sunitha"
#define STAPSK  "1234567890"
#endif

const char* ssid = STASSID;
const char* password = STAPSK;

void OTAsetup() {
  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password);
  while (WiFi.waitForConnectResult() != WL_CONNECTED) {
    delay(5000);
    ESP.restart();
  }
  ArduinoOTA.begin();
}

void OTAloop() {
  ArduinoOTA.handle();
}

void setup() {
  OTAsetup();
  
  // Custom setup code from code-1
  pinMode(2, OUTPUT);
  pinMode(3, INPUT_PULLUP); // Assuming pin 3 is connected to the positive rail
  pinMode(4, INPUT_PULLUP); // Assuming pin 4 is connected to the negative rail
  pinMode(5, INPUT_PULLUP);
}

void loop() {
  OTAloop();
  delay(10);  // If no custom loop code, ensure to have a delay in loop

  // Custom loop code from code-1
  bool X = digitalRead(3); // Read the state of pin 3 (positive rail)
  bool Y = digitalRead(4); // Read the state of pin 4 (negative rail)
  bool Z = digitalRead(5); // Read the state of pin 5 (additional input for Z)
  bool R1 = (!X && !Z) || (Y && !Z);
  if (R1 == 1) {
    digitalWrite(2, HIGH);
    delay(1000);
  } else {
    digitalWrite(2, LOW);
    delay(1000);
  }
}
