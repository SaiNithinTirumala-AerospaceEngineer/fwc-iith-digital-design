// Define the digital pins for LEDs
const int LED_X = 2; // Assuming X is connected to pin 2
const int LED_Y = 3; // Assuming Y is connected to pin 3
const int LED_Z = 4; // Assuming Z is connected to pin 4
const int LED_F = 5; // Output LED connected to pin 5

void setup(){
  OTAsetup();

  //-------------------//
  // Custom setup code //
  //-------------------//
  pinMode(LED_X, INPUT_PULLUP); // Using internal pull-up resistors
  pinMode(LED_Y, INPUT_PULLUP);
  pinMode(LED_Z, INPUT_PULLUP);
  pinMode(LED_F, OUTPUT);
}

void loop() {
  OTAloop();
  delay(10);  // If no custom loop code ensure to have a delay in loop

  //-------------------//
  // Custom loop code  //
  //-------------------//
  // Read the inputs from jumpers connected to power rails
  int X = digitalRead(LED_X);
  int Y = digitalRead(LED_Y);
  int Z = digitalRead(LED_Z);

  // Evaluate the boolean expression
  int F = (!X && !Z) || (Y && Z);

  // Display the output on LED connected to pin 5
  digitalWrite(LED_F, F);

  // Delay for stability
  delay(100);
}
