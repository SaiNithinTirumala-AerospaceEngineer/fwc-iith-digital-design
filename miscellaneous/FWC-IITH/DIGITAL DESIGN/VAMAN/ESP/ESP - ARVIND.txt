// Define the pins for input variables
const int pinX = 2;
const int pinY = 3;
const int pinZ = 4;
const int pinW = 5;

// Define the pin for the output LED
const int outputPin = 13;

void setup() {
  // Initialize the input pins
  pinMode(pinX, INPUT);
  pinMode(pinY, INPUT);
  pinMode(pinZ, INPUT);
  pinMode(pinW, INPUT);
  
  // Initialize the output pin
  pinMode(outputPin, OUTPUT);
}

void loop() {
  // Read the input values
  int X = digitalRead(pinX);
  int Y = digitalRead(pinY);
  int Z = digitalRead(pinZ);
  int W = digitalRead(pinW);
  
  // Evaluate the boolean expression F=X'Y'Z'W+YZ+XZ'W
  int F = (!X && !Y && !Z && !W) || (Y && Z) || (X && !Z && W);
  
  // Light up the LED if the evaluated value is 1
  if (F == 1) {
    digitalWrite(outputPin, HIGH);
  } else {
    digitalWrite(outputPin, LOW);
  }

  // Delay for stability
  delay(100);
}
